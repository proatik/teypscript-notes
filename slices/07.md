## 7. Generics

### ✅ What are Generics

Generics let you write **reusable, type-safe code** that works with multiple types without sacrificing type information. Instead of using `any`, you use a **type parameter** that is resolved at call time.

```ts
// Without generics — loses type info
function identity(value: any): any {
  return value;
}

// With generics — preserves type info
function identity<T>(value: T): T {
  return value;
}

identity("hello"); // returns string
identity(42);      // returns number
identity(true);    // returns boolean
```

`T` is a convention for the first type parameter. You can use any name (`U`, `K`, `V`, `Item`, etc.).


### ✅ Generic Functions

TypeScript usually **infers** the type argument from the call — no need to write `<string>` explicitly.

```ts
function wrap<T>(value: T): { value: T } {
  return { value };
}

const a = wrap("hello");  // { value: string }
const b = wrap(42);       // { value: number }

// Explicit type argument when inference fails
const c = wrap<boolean>(true);
```

#### Multiple type parameters

```ts
function pair<K, V>(key: K, value: V): [K, V] {
  return [key, value];
}

pair("name", "Atik");   // [string, string]
pair("age", 30);        // [string, number]
```

#### Generic arrow functions

```ts
const identity = <T>(value: T): T => value;

// In .tsx files, use trailing comma to avoid JSX ambiguity
const identity = <T,>(value: T): T => value;
```


### ✅ Generic Interfaces & Types

```ts
interface ApiResponse<T> {
  data: T;
  status: number;
  message: string;
}

type Result<T, E = Error> = { ok: true; value: T } | { ok: false; error: E };
```

```ts
const response: ApiResponse<User[]> = {
  data: [{ name: "Atik", age: 30 }],
  status: 200,
  message: "ok",
};
```

#### Generic classes

```ts
class Stack<T> {
  private items: T[] = [];

  push(item: T): void {
    this.items.push(item);
  }

  pop(): T | undefined {
    return this.items.pop();
  }

  peek(): T | undefined {
    return this.items[this.items.length - 1];
  }
}

const numStack = new Stack<number>();
numStack.push(1);
numStack.push(2);
numStack.pop(); // number
```


### ✅ Generic Constraints (`extends`)

Constrain a type parameter to a subset of types — ensures the type has certain properties.

```ts
// T must have a .length property
function getLength<T extends { length: number }>(value: T): number {
  return value.length;
}

getLength("hello");    // 5 — string has length
getLength([1, 2, 3]);  // 3 — array has length
getLength(42);         // Error — number has no length
```

#### Constraining to object keys

```ts
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}

const user = { name: "Atik", age: 30 };

getProperty(user, "name"); // string
getProperty(user, "age");  // number
getProperty(user, "foo");  // Error — "foo" is not keyof user
```

`keyof T` produces a union of all keys of `T`. `T[K]` is the type of the value at key `K`.


### ✅ Default Type Parameters

Provide a fallback type when no argument is given.

```ts
interface Container<T = string> {
  value: T;
}

const a: Container = { value: "hello" };      // T defaults to string
const b: Container<number> = { value: 42 };   // T is number
```

Useful in library APIs where you want a sensible default but allow customisation.


### ✅ `const` Type Parameters (TS 5.0+)

Without `const`, TypeScript **widens** inferred types (e.g. `"left"` becomes `string`). The `const` modifier forces TypeScript to infer the **narrowest literal type** — equivalent to `as const` but applied at the call site automatically.

```ts
// Without const — type is widened
function makeConfig<T extends object>(config: T) {
  return config;
}

const c1 = makeConfig({ port: 3000, env: "prod" });
// c1: { port: number; env: string } — too wide

// With const — type is preserved as literals
function makeConfig<const T extends object>(config: T) {
  return config;
}

const c2 = makeConfig({ port: 3000, env: "prod" });
// c2: { readonly port: 3000; readonly env: "prod" } — literal types preserved
```

This is especially useful for builder patterns, route definitions, and config objects where specific values matter.

```ts
function buildRoute<const T extends string>(path: T) {
  return { path, method: "GET" as const };
}

const route = buildRoute("/api/users");
// route: { path: "/api/users"; method: "GET" }
// Without const: { path: string; method: "GET" }
```


### ✅ Generic Utility Patterns

#### Extract array element type

```ts
type ElementOf<T> = T extends (infer E)[] ? E : never;

type A = ElementOf<string[]>;  // string
type B = ElementOf<number[]>;  // number
```

#### Extract promise resolved type

```ts
type Awaited<T> = T extends Promise<infer R> ? R : T;

type C = Awaited<Promise<string>>;  // string
```

#### Conditional default with generic

```ts
function createRequest<T = void>(url: string): Promise<T> {
  return fetch(url).then(r => r.json());
}

createRequest("/api/users");            // Promise<void>
createRequest<User[]>("/api/users");    // Promise<User[]>
```
