## 4. Type Narrowing & Type Guards

### ✅ What is Narrowing

TypeScript starts with a wide type (e.g. `string | number`) and **narrows** it to a more specific type inside a conditional block. The narrowed type is only valid within that block.

```ts
function process(value: string | number) {
  // value is string | number here

  if (typeof value === "string") {
    // value is string here
    console.log(value.toUpperCase());
  } else {
    // value is number here
    console.log(value.toFixed(2));
  }
}
```


### ✅ `typeof` Guard

Works for JavaScript primitives: `string`, `number`, `bigint`, `boolean`, `symbol`, `undefined`, `function`, `object`.

```ts
function format(x: string | number | boolean): string {
  if (typeof x === "string")  return x.toUpperCase();
  if (typeof x === "number")  return x.toFixed(2);
  if (typeof x === "boolean") return x ? "yes" : "no";
  return x; // never — all cases handled
}
```

**Gotcha:** `typeof null === "object"` — always check for `null` separately.

```ts
function process(val: string | null) {
  if (typeof val === "object") {
    // val could be null here — check!
  }
  if (val !== null && typeof val === "object") {
    // safe
  }
}
```


### ✅ `instanceof` Guard

Narrows to a specific class instance.

```ts
class Dog { bark() { return "woof"; } }
class Cat { meow() { return "meow"; } }

function speak(animal: Dog | Cat) {
  if (animal instanceof Dog) {
    return animal.bark(); // Dog here
  }
  return animal.meow();   // Cat here
}
```

Works with any class, including built-ins (`Date`, `Error`, `Map`, etc.):

```ts
function handle(value: Date | string) {
  if (value instanceof Date) {
    return value.toISOString(); // Date here
  }
  return value.trim(); // string here
}
```


### ✅ `in` Guard

Checks if a property exists on an object — useful for narrowing object union types.

```ts
type Fish = { swim(): void };
type Bird = { fly(): void };

function move(animal: Fish | Bird) {
  if ("swim" in animal) {
    animal.swim(); // Fish here
  } else {
    animal.fly();  // Bird here
  }
}
```

Also works with optional properties:

```ts
type Admin = { role: "admin"; permissions: string[] };
type User  = { role: "user" };

function check(person: Admin | User) {
  if ("permissions" in person) {
    console.log(person.permissions); // Admin here
  }
}
```


### ✅ Custom Type Guards (`is`)

A function that returns a **type predicate** — tells TypeScript what type the argument is if the function returns `true`.

```ts
function isString(value: unknown): value is string {
  return typeof value === "string";
}

function process(value: unknown) {
  if (isString(value)) {
    console.log(value.toUpperCase()); // value is string here
  }
}
```

Syntax: `parameterName is Type` as the return type. The function must return a `boolean`.

```ts
function isError(value: unknown): value is Error {
  return value instanceof Error;
}

function isUser(obj: unknown): obj is { name: string; age: number } {
  return (
    typeof obj === "object" &&
    obj !== null &&
    "name" in obj &&
    "age" in obj
  );
}
```

#### Inferred type predicates (TS 5.5+)

TypeScript 5.5 can **automatically infer** the type predicate without you writing `value is T` — if the function has a single simple return expression tied to a type check.

```ts
// Before TS 5.5 — manual predicate required
const isNonNull = <T>(val: T | null): val is T => val !== null;

// TS 5.5+ — inferred automatically
const isNonNull = <T>(val: T | null) => val !== null;

const users: (string | null)[] = ["Atik", null, "Bob"];

// TS 5.5+: filter correctly infers string[] without manual predicate
const names = users.filter(u => u !== null); // string[]
```

This eliminates a large class of boilerplate type guard functions.


### ✅ Discriminated Unions & Exhaustiveness

When a union uses a **discriminant** (shared literal property), TypeScript narrows automatically inside each branch.

```ts
type Action =
  | { type: "increment"; amount: number }
  | { type: "decrement"; amount: number }
  | { type: "reset" };

function reducer(state: number, action: Action): number {
  switch (action.type) {
    case "increment": return state + action.amount;
    case "decrement": return state - action.amount;
    case "reset":     return 0;
    default:
      const _check: never = action; // compile error if a new Action type is added
      return state;
  }
}
```

#### Assertion function pattern

An alternative to type predicates — throws if the condition fails, narrows the type if it passes.

```ts
function assertIsString(val: unknown): asserts val is string {
  if (typeof val !== "string") {
    throw new Error(`Expected string, got ${typeof val}`);
  }
}

function process(val: unknown) {
  assertIsString(val);
  // val is string from here on — no if block needed
  console.log(val.toUpperCase());
}
```
